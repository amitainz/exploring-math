\newcommand{\celltitle}[2]{%
  {\setlength{\fboxsep}{1.8mm}%
  \colorbox{#1!11}{%
    \parbox{\dimexpr\linewidth-2\fboxsep\relax}{%
      \textcolor{#1!70!black}{\bfseries #2}%
    }%
  }}%
  \par\vspace{1.6mm}%
}

\newcommand{\worksheetcell}[3]{%
  \fcolorbox{cardborder}{cardfill}{%
    \begin{minipage}[t][\cellheight][t]{\dimexpr\cellwidth-2\fboxrule\relax}
      \setlength{\parindent}{0pt}%
      \setlength{\parskip}{0pt}%
      \vspace{2mm}%
      \hspace*{2.3mm}%
      \begin{minipage}[t]{\dimexpr\linewidth-4.6mm\relax}
        \celltitle{#1}{#2}%

      \end{minipage}%
    \end{minipage}%
  }%
}

\newcommand{\drawFence}[1]{\draw[fence] #1; \draw[fencelight] #1;}

\newcommand{\diagramTriangle}{%
  \begin{center}
  \begin{tikzpicture}[scale=0.88]
    \coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (1.6,2.5);
    \fill[greenregion] (A)--(B)--(C)--cycle;
    \drawFence{(A)--(B)--(C)--cycle}
    \foreach \p in {A,B,C} \node[post] at (\p) {};
  \end{tikzpicture}
  \end{center}
}

\newcommand{\diagramFourPosts}{%
  \begin{center}
  \begin{tikzpicture}[scale=0.78]
    \coordinate (A) at (0,0); \coordinate (B) at (3.1,0); \coordinate (C) at (3.1,2.2); \coordinate (D) at (0,2.2);
    \fill[greenregion] (A)--(B)--(C)--cycle;
    \fill[blueregion] (A)--(C)--(D)--cycle;
    \drawFence{(A)--(B)--(C)--(D)--cycle}
    \drawFence{(A)--(C)}
    \foreach \p in {A,B,C,D} \node[post] at (\p) {};
  \end{tikzpicture}
  \end{center}
}

\newcommand{\diagramTree}{%
  \begin{center}
  \begin{tikzpicture}[scale=0.9]
    \coordinate (O) at (1.6,1.2); \coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (1.6,2.7);
    \drawFence{(O)--(A)}
    \drawFence{(O)--(B)}
    \drawFence{(O)--(C)}
    \foreach \p in {O,A,B,C} \node[post] at (\p) {};
  \end{tikzpicture}
  \end{center}
}

\newcommand{\diagramFourRegions}{%
  \begin{center}
  \begin{tikzpicture}[scale=0.78]
    \coordinate (A) at (0,0); \coordinate (B) at (3,0); \coordinate (C) at (3,2.5); \coordinate (D) at (0,2.5); \coordinate (O) at (1.5,1.25);
    \fill[greenregion] (A)--(B)--(O)--cycle;
    \fill[blueregion] (B)--(C)--(O)--cycle;
    \fill[yellowregion] (C)--(D)--(O)--cycle;
    \fill[pinkregion] (D)--(A)--(O)--cycle;
    \drawFence{(A)--(B)--(C)--(D)--cycle}
    \drawFence{(A)--(O)}
    \drawFence{(B)--(O)}
    \drawFence{(C)--(O)}
    \drawFence{(D)--(O)}
    \foreach \p in {A,B,C,D,O} \node[post] at (\p) {};
  \end{tikzpicture}
  \end{center}
}
