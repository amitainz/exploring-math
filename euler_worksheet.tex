\newcommand{\drawFence}[1]{\draw[fence] #1; \draw[fencelight] #1;}
\newcommand{\diagramLabel}[1]{\node[font=\bfseries\large] at (-0.65,2.95) {#1};}

\newcommand{\diagramA}{%
  \begin{tikzpicture}[scale=0.66]
    \coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (1.6,2.45);
    \fill[pattern=north east lines,pattern color=black] (A)--(B)--(C)--cycle;
    \drawFence{(A)--(B)--(C)--cycle}
    \foreach \p in {A,B,C} \node[post] at (\p) {};
    \diagramLabel{A}
  \end{tikzpicture}%
}

\newcommand{\diagramB}{%
  \begin{tikzpicture}[scale=0.66]
    \coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (3.2,2.25); \coordinate (D) at (0,2.25);
    \fill[pattern=north west lines,pattern color=black] (A)--(B)--(C)--cycle;
    \fill[pattern=dots,pattern color=black] (A)--(C)--(D)--cycle;
    \drawFence{(A)--(B)--(C)--(D)--cycle}
    \drawFence{(A)--(C)}
    \foreach \p in {A,B,C,D} \node[post] at (\p) {};
    \diagramLabel{B}
  \end{tikzpicture}%
}

\newcommand{\diagramC}{%
  \begin{tikzpicture}[scale=0.7]
    \coordinate (O) at (1.6,1.05); \coordinate (A) at (0,0); \coordinate (B) at (3.2,0); \coordinate (C) at (1.6,2.55);
    \drawFence{(O)--(A)}
    \drawFence{(O)--(B)}
    \drawFence{(O)--(C)}
    \foreach \p in {O,A,B,C} \node[post] at (\p) {};
    \diagramLabel{C}
  \end{tikzpicture}%
}

\newcommand{\diagramD}{%
  \begin{tikzpicture}[scale=0.62]
    \coordinate (A) at (0,0); \coordinate (B) at (3.4,0); \coordinate (C) at (3.4,2.65); \coordinate (D) at (0,2.65); \coordinate (O) at (1.7,1.3);
    \fill[pattern=north east lines,pattern color=black] (A)--(B)--(O)--cycle;
    \fill[pattern=north west lines,pattern color=black] (B)--(C)--(O)--cycle;
    \fill[pattern=dots,pattern color=black] (C)--(D)--(O)--cycle;
    \fill[pattern=horizontal lines,pattern color=black] (D)--(A)--(O)--cycle;
    \drawFence{(A)--(B)--(C)--(D)--cycle}
    \drawFence{(A)--(O)}
    \drawFence{(B)--(O)}
    \drawFence{(C)--(O)}
    \drawFence{(D)--(O)}
    \foreach \p in {A,B,C,D,O} \node[post] at (\p) {};
    \diagramLabel{D}
  \end{tikzpicture}%
}
